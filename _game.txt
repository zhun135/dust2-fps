// COLLISION
function getBox(m){m.updateWorldMatrix(false,false);return new THREE.Box3().setFromObject(m)}
function collides(cx,cy,cz,boxes){for(const b of boxes){if(cx+R<=b.min.x||cx-R>=b.max.x)continue;if(cz+R<=b.min.z||cz-R>=b.max.z)continue;if(cy+HH<=b.min.y||cy-HH>=b.max.y)continue;return true}return false}
